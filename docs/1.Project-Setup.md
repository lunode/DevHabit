```xml
    <CodeAnalysisTreatWarningsAsErrors>true</CodeAnalysisTreatWarningsAsErrors>
```

```xml
</Project>
```

## Containerization, Docker compose

DevHabit.API/Dockerfile.

```yml
# See https://aka.ms/customizecontainer to learn how to customize your debug container and how Visual Studio uses this

# This stage is used when running from VS in fast mode (Default for Debug configuration)
FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS base
USER $APP_UID
WORKDIR /app
EXPOSE 8080
EXPOSE 8081


# This stage is used to build the service project
FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build
ARG BUILD_CONFIGURATION=Release
WORKDIR /src
COPY ["Directory.Packages.props", "."]
COPY ["Directory.Build.props", "."]
COPY ["DevHabit.Api/DevHabit.Api.csproj", "DevHabit.Api/"]
RUN dotnet restore "./DevHabit.Api/DevHabit.Api.csproj"
COPY . .
WORKDIR "/src/DevHabit.Api"
RUN dotnet build "./DevHabit.Api.csproj" -c $BUILD_CONFIGURATION -o /app/build


# This stage is used to publish the service project to be copied to the final stage
FROM build AS publish
ARG BUILD_CONFIGURATION=Release
RUN dotnet publish "./DevHabit.Api.csproj" -c $BUILD_CONFIGURATION -o /app/publish /p:UseAppHost=false


# This stage is used in production or when running from VS in regular mode (Default when not using the Debug configuration)
FROM base AS final
WORKDIR /app
COPY --from=publish /app/publish .
ENTRYPOINT ["dotnet", "DevHabit.Api.dll"]
```

## OpenTelemetry, Seq, Aspire Dashboard

```sh

dotnet add package OpenTelemetry.Extensions.Hosting
dotnet add package OpenTelemetry.Exporter.OpenTelemetryProtocol
dotnet add package OpenTelemetry.Instrumentation.Http
dotnet add package OpenTelemetry.Instrumentation.AspNetCore
dotnet add package OpenTelemetry.Instrumentation.Runtime
```

DevHabit.API/Program.cs

```cs
builder.Services.AddOpenTelemetry()
    .ConfigureResource(resource => resource.AddService(builder.Environment.ApplicationName))
    .WithTracing(trace => trace
        .AddHttpClientInstrumentation()
        .AddAspNetCoreInstrumentation())
    .WithMetrics(metric => metric
        .AddHttpClientInstrumentation()
        .AddAspNetCoreInstrumentation()
        .AddRuntimeInstrumentation())
    .UseOtlpExporter();

builder.Logging.AddOpenTelemetry(options =>
{
    options.IncludeScopes = true; // default to false
    options.IncludeFormattedMessage = true;
});
```

## CI/CD, Github Actions

.github/workflows/build.yml

```yml
name: Build

on:
  push:
    branches: [master]
  pull_request:
    branches: [master]

env:
  DOTNET_VERSION: "10.x"
jobs:
  build:
    runs-on: ubuntu-latest

    defaults:
      run:
        working-directory: .

    steps:
      - uses: actions/checkout@v4

      - name: Setup .NET
        uses: actions/setup-dotnet@v4
        with:
          dotnet-version: ${{ env.DOTNET_VERSION }}

      - name: Restore
        run: dotnet restore DevHabit.slnx

      - name: Build
        run: dotnet build DevHabit.slnx --configuration Release --no-restore

      - name: Test
        # run: dotnet test DevHabit.slnx --no-build --verbosity normal --logger trx --results-directory TestResults
        run: dotnet test DevHabit.slnx --configuration Release --no-build --no-restore

      - name: Publish
        run: dotnet publish DevHabit.slnx --configuration Release --no-build --no-restore
```
