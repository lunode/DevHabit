```cs
public sealed class DataShapingService
{
  //...
  public List<ExpandoObject> ShapeCollectionData<T>(
      IEnumerable<T> entities,
      string? fields,
      Func<T, List<LinkDto>>? linksFactory = null)
  {
      var fieldSets = GetFieldSets(fields);
      PropertyInfo[] propertyInfos = GetProperties(typeof(T));

      if (fieldSets.Any())
      {
          propertyInfos = propertyInfos
              .Where(p => fieldSets.Contains(p.Name, StringComparer.OrdinalIgnoreCase))
              .ToArray();
      }

      List<ExpandoObject> shapedObjects = [];
      foreach (var entity in entities)
      {
          IDictionary<string, object?> shapedObject = new ExpandoObject();
          foreach (var propertyInfo in propertyInfos)
          {
              shapedObject[propertyInfo.Name] = propertyInfo.GetValue(entity);
          }
          if (linksFactory is not null)
          {
              shapedObject["links"] = linksFactory(entity);
          }
          shapedObjects.Add((ExpandoObject)shapedObject);
      }
      return shapedObjects;
  }
}
```

```cs
private List<LinkDto> CreateLinksForHabits(HabitsQueryParameters queryParameters, bool hasPreviousPage, bool hasNextPage)
{
    List<LinkDto> links =
    [
        linkService.Create(nameof(GetHabits), "self", HttpMethods.Get, new {
            page = queryParameters.Page,
            pageSize = queryParameters.PageSize,
            q = queryParameters.Search,
            sort = queryParameters.Sort,
            status = queryParameters.Status,
            type = queryParameters.Type,
            fields = queryParameters.Fields,
        }),
        linkService.Create(nameof(CreateHabit), "create", HttpMethods.Post)
    ];
    if (hasPreviousPage)
    {
        links.Add(
            linkService.Create(nameof(GetHabits), "previous-page", HttpMethods.Get, new
            {
                page = queryParameters.Page - 1,
                pageSize = queryParameters.PageSize,
                q = queryParameters.Search,
                sort = queryParameters.Sort,
                status = queryParameters.Status,
                type = queryParameters.Type,
                fields = queryParameters.Fields,
            }),
        );
    }
    if (hasNextPage)
    {
        links.Add(
            linkService.Create(nameof(GetHabits), "next-page", HttpMethods.Get, new
            {
                page = queryParameters.Page + 1,
                pageSize = queryParameters.PageSize,
                q = queryParameters.Search,
                sort = queryParameters.Sort,
                status = queryParameters.Status,
                type = queryParameters.Type,
                fields = queryParameters.Fields,
            })
        );
    }
    return links;
}
```

```cs
                new MediaTypeApiVersionReaderBuilder()
                    .Template(CustomMediaTypeNames.Application.HateoasJsonTemplate)
                    .Build()
                );
        })
```
